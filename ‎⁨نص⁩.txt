    <script>
        let currentDate = new Date();
        let currentChalet = 'الرئيسي';
        let chalets = JSON.parse(localStorage.getItem('alghroob_chalets')) || ['الرئيسي'];
        let bookings = JSON.parse(localStorage.getItem('alghroob_bookings')) || {};
        let activeDateKey = null;

        function initApp() {
            const select = document.getElementById('chaletSelect');
            select.innerHTML = chalets.map(c => `<option value="${c}">${c === 'الرئيسي' ? 'شاليه الغروب الرئيسي' : c}</option>`).join('');
            select.value = currentChalet;
            loadLogo();
            renderCalendar();
        }

        function updateLogo(event) {
            const file = event.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    localStorage.setItem('alghroob_logo', e.target.result);
                    loadLogo();
                }
                reader.readAsDataURL(file);
            }
        }

        function loadLogo() {
            const savedLogo = localStorage.getItem('alghroob_logo');
            const container = document.getElementById('logoContainer');
            if (savedLogo) {
                container.innerHTML = `<img src="${savedLogo}" class="w-full h-full object-cover rounded-xl">`;
            }
        }

        function switchChalet(val) {
            currentChalet = val;
            renderCalendar();
        }

        function addNewChalet() {
            const name = prompt('أدخل اسم الشاليه الجديد:');
            if (name && name.trim() !== '') {
                if (!chalets.includes(name.trim())) {
                    chalets.push(name.trim());
                    localStorage.setItem('alghroob_chalets', JSON.stringify(chalets));
                    currentChalet = name.trim();
                    initApp();
                }
            }
        }

        function renderCalendar() {
            const year = currentDate.getFullYear();
            const month = currentDate.getMonth();
            const monthsNames = ["يناير", "فبراير", "مارس", "أبريل", "مايو", "يونيو", "يوليو", "أغسطس", "سبتمبر", "أكتوبر", "نوفمبر", "ديسمبر"];
            document.getElementById('calendarMonthYear').innerText = `${monthsNames[month]} ${year}`;

            const firstDayIndex = new Date(year, month, 1).getDay();
            const totalDays = new Date(year, month + 1, 0).getDate();
            const daysContainer = document.getElementById('calendarDays');
            daysContainer.innerHTML = '';

            let totalBookingsCount = 0;
            let totalPaidSum = 0;
            let totalRemainingSum = 0;

            for (let i = 0; i < firstDayIndex; i++) {
                daysContainer.innerHTML += `<div class="h-24 bg-transparent"></div>`;
            }

            for (let day = 1; day <= totalDays; day++) {
                const dateKey = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
                const chaletBookings = bookings[currentChalet] || {};
                const booking = chaletBookings[dateKey];

                let bgClass = "bg-[#161F38]/60 border-[#3A506B] hover:border-[#5BC0BE]";
                let badgeText = "متاح";
                let badgeColor = "text-emerald-400 bg-emerald-500/10";

                if (booking) {
                    totalBookingsCount++;
                    totalPaidSum += Number(booking.paid || 0);
                    totalRemainingSum += Number(booking.remaining || 0);
                    bgClass = "bg-[#1C2541] border-[#5BC0BE] shadow-md";
                    badgeText = booking.section;
                    badgeColor = "text-[#5BC0BE] bg-[#5BC0BE]/20";
                }

                daysContainer.innerHTML += `
                    <div onclick="openDayModal('${dateKey}')" class="h-24 rounded-xl border ${bgClass} p-2 flex flex-col justify-between cursor-pointer transition transform active:scale-95">
                        <div class="flex justify-between items-center">
                            <span class="text-sm font-bold text-white">${day}</span>
                            <span class="text-[10px] px-1.5 py-0.5 rounded font-medium ${badgeColor}">${badgeText}</span>
                        </div>
                        ${booking ? `<div class="text-[11px] text-gray-300 truncate font-medium"><i class="fa-solid fa-user text-[9px] ml-1"></i>${booking.name}</div><div class="text-[10px] text-amber-400 font-mono">باقي: ${booking.remaining} ر.س</div>` : `<div class="text-[10px] text-gray-500">انقر للحجز</div>`}
                    </div>
                `;
            }

            document.getElementById('statTotalBookings').innerText = totalBookingsCount;
            document.getElementById('statTotalPaid').innerText = totalPaidSum + ' ر.س';
            document.getElementById('statTotalRemaining').innerText = totalRemainingSum + ' ر.س';
        }

        function changeMonth(dir) {
            currentDate.setMonth(currentDate.getMonth() + dir);
            renderCalendar();
        }

        function openDayModal(dateKey) {
            activeDateKey = dateKey;
            document.getElementById('selectedDateInput').value = dateKey;
            document.getElementById('bookingModal').classList.remove('hidden');
            
            const chaletBookings = bookings[currentChalet] || {};
            const booking = chaletBookings[dateKey];

            if (booking) {
                document.getElementById('modalTitle').innerText = `تفاصيل حجز تاريخ: ${dateKey}`;
                document.getElementById('viewMode').classList.remove('hidden');
                document.getElementById('bookingForm').classList.add('hidden');

                document.getElementById('viewName').innerText = booking.name;
                document.getElementById('viewPhone').innerText = booking.phone;
                document.getElementById('viewSection').innerText = booking.section;
                document.getElementById('viewTotal').innerText = booking.total + ' ر.س';
                document.getElementById('viewPaid').innerText = booking.paid + ' ر.س';
                document.getElementById('viewRemaining').innerText = booking.remaining + ' ر.س';
            } else {
                document.getElementById('modalTitle').innerText = `تسجيل حجز جديد (${dateKey})`;
                document.getElementById('viewMode').classList.add('hidden');
                document.getElementById('bookingForm').classList.remove('hidden');
                document.getElementById('bookingForm').reset();
            }
        }

        function closeModal() {
            document.getElementById('bookingModal').classList.add('hidden');
        }

        function calcRemaining() {
            const total = parseFloat(document.getElementById('bookingTotal').value) || 0;
            const paid = parseFloat(document.getElementById('bookingPaid').value) || 0;
            document.getElementById('bookingRemaining').value = Math.max(0, total - paid);
        }

        function saveBooking(e) {
            e.preventDefault();
            if (!bookings[currentChalet]) bookings[currentChalet] = {};

            bookings[currentChalet][activeDateKey] = {
                name: document.getElementById('clientName').value,
                phone: document.getElementById('clientPhone').value,
                section: document.getElementById('bookingSection').value,
                total: document.getElementById('bookingTotal').value,
                paid: document.getElementById('bookingPaid').value,
                remaining: document.getElementById('bookingRemaining').value
            };

            localStorage.setItem('alghroob_bookings', JSON.stringify(bookings));
            closeModal();
            renderCalendar();
        }

        function deleteBookingConfirm() {
            if (confirm('هل أنت متأكد من حذف هذا الحجز؟')) {
                if (bookings[currentChalet]) {
                    delete bookings[currentChalet][activeDateKey];
                    localStorage.setItem('alghroob_bookings', JSON.stringify(bookings));
                }
                closeModal();
                renderCalendar();
            }
        }

        function sendWhatsApp() {
            const chaletBookings = bookings[currentChalet] || {};
            const b = chaletBookings[activeDateKey];
            if (b) {
                const msg = `مرحباً ${b.name}، نؤكد لك حجزك في شاليه الغروب (${currentChalet}) ليوم ${activeDateKey} (${b.section}). المبلغ الإجمالي: ${b.total} ر.س، المدفوع: ${b.paid} ر.س، المتبقي: ${b.remaining} ر.س. نسعد بخدمتكم!`;
                window.open(`https://wa.me/${b.phone}?text=${encodeURIComponent(msg)}`, '_blank');
            }
        }

        function openSettings() { document.getElementById('settingsModal').classList.remove('hidden'); }
        function closeSettings() { document.getElementById('settingsModal').classList.add('hidden'); }

        function exportData() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(bookings));
            const downloadAnchor = document.createElement('a');
            downloadAnchor.setAttribute("href", dataStr);
            downloadAnchor.setAttribute("download", "alghroob_backup.json");
            document.body.appendChild(downloadAnchor);
            downloadAnchor.click();
            downloadAnchor.remove();
        }

        function importData(event) {
            const file = event.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    try {
                        bookings = JSON.parse(e.target.result);
                        localStorage.setItem('alghroob_bookings', JSON.stringify(bookings));
                        renderCalendar();
                        closeSettings();
                        alert('تم استيراد البيانات بنجاح!');
                    } catch (err) {
                        alert('ملف غير صالح!');
                    }
                }
                reader.readAsText(file);
            }
        }

        window.onload = initApp;
    </script>
